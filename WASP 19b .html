# 1. Strip unit wrappers using pure numbers to prevent any errors
time_vals = np.array(lc.time.value, dtype=float)
period_val = float(best_period.value)
t0_val = float(best_t0.value)

# 2. Create a mask hiding everything within ~1.2 hours of each transit center
phase = np.abs((time_vals - t0_val + 0.5 * period_val) % period_val - 0.5 * period_val)
planet_mask = phase < 0.05

# 3. Re-flatten ONLY the starlight outside the masked transits
true_flat_lc = lc.flatten(window_length=101, mask=planet_mask).remove_outliers(sigma=5)

# 4. Re-measure the true, un-flattened dip depth
true_bls = true_flat_lc.to_periodogram(method='bls', period=period_grid)
true_depth = float(true_bls.depth_at_max_power.value)

# 5. Convert depth to Jupiter radii (WASP-19 star is ~9.73 Jupiter radii)
true_radius = np.sqrt(true_depth) * 9.73

print(f"True Transit Depth: {true_depth * 100:.2f}% of starlight blocked")
print(f"Calculated Planet Size: {true_radius:.2f} times the radius of Jupiter!")

# 6. Stack (phase-fold) all transits and plot the final proof
folded_lc = true_flat_lc.fold(period=best_period, epoch_time=best_t0)
fig, ax = plt.subplots(figsize=(8, 5))
folded_lc.scatter(ax=ax, color='blue', s=2)
plt.title(f"Final Phase-Folded Transit of WASP-19b (Radius: {true_radius:.2f} R_Jup)")
plt.xlim(-0.2, 0.2)
plt.show()

# 1. Create a binned (averaged) version of your folded light curve for a crisp overlay
binned_lc = folded_lc.bin(time_bin_size=0.005)

# 2. Calculate orbital distance using Kepler's Third Law
# WASP-19 is a Sun-like star (~0.904 solar masses), and 1 AU = Earth's distance from the Sun
period_years = float(best_period.value) / 365.25
star_mass_solar = 0.904
orbital_distance_au = (star_mass_solar * (period_years ** 2)) ** (1/3)

# Convert AU to kilometers (1 AU = 149.6 million km)
orbital_distance_km = orbital_distance_au * 149_600_000

print(f"Orbital Distance: {orbital_distance_au:.4f} AU ({orbital_distance_km:,.0f} km)")
print(f"Comparison: Earth orbits at 1.0000 AU — WASP-19b is ~{1/orbital_distance_au:.0f}x closer to its sun!")

# 3. Generate a publication-ready final figure combining raw points + binned curve
fig, ax = plt.subplots(figsize=(9, 5))
folded_lc.scatter(ax=ax, color='royalblue', alpha=0.3, s=3, label='Individual TESS Measurements')
binned_lc.plot(ax=ax, color='crimson', linewidth=2.5, label='Binned Transit Profile')

plt.title(f"WASP-19b Transit Profile | Radius: {true_radius:.2f} R_Jup | Orbit: {orbital_distance_au:.4f} AU")
plt.xlabel("Phase (Days from Transit Center)")
plt.ylabel("Normalized Starlight Flux")
plt.xlim(-0.15, 0.15)
plt.legend(loc='lower right')
plt.grid(alpha=0.2)
plt.show()
